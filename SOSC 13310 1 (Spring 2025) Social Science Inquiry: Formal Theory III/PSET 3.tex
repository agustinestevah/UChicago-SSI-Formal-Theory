    \section*{Question 1 (8 points)}
     In the cheap-talk model with $b > 0$, consider $M = \{G, I, B\}$. Assume $\theta$ is uniformly distributed on $[0, 1]$.
    \begin{enumerate}
        \item (2 points)
        Assume the politician chooses $x_m$ corresponding to each message $m \in M$ and $0 < x_G < x_I < x_B < 1$. What is the expert’s optimal message for each $\theta \in [0, 1]$?
        \begin{solution}
        There are three cases
            \begin{itemize}
                \item The expert will choose $B$ if and only if
                \[(x_B - (\theta + b))^2 \leq \min\{(x_G - (\theta + b))^2, (x_I - (\theta + b))^2\}\] solving both of these equations (using the solution from class), we get that 
                \begin{align*}
                    \theta \geq \frac{x_B + x_G}{2} - b\\
                    \theta \geq \frac{x_B + x_I}{2} - b
                \end{align*}
                Thus, the expert will choose $B$ iff
                \begin{align}
                    \boxed{\theta \geq \max\{\frac{x_B + x_G}{2} - b, \frac{x_B + x_I}{2} - b\} = \max\{\frac{x_B + x_I}{2} - b,0\} =: k'}
                 \end{align}
                \item The expert will choose $G$ if and only if 
                \[(x_G - (\theta + b))^2 \leq \min\{(x_I - (\theta + b))^2, (x_G - (\theta + b))^2\}\]
                Solving, we see that 
                \begin{align*}
                    2(x_G - x_I)(\frac{x_G + x_I}{2} - (\theta + b)) &\leq 0\\ 
                    &\iff\\ 
                    2(x_I- x_G)(\frac{x_G + x_I}{2} - (\theta + b)) &\geq 0 \\
                    &\iff\\
                    \theta &\leq \frac{x_G + x_I}{2} - b
                \end{align*}
                Similarly, 
                \[\theta \leq \frac{x_G + x_B}{2} - b\] Thus, the expert will choose $G$ iff 
                \begin{align}
                    \boxed{\theta \leq \min\{\frac{x_B + x_G}{2} - b, \frac{x_G +x_I}{2} - b\} = \max\{\frac{x_I + x_G}{2}- b,0\}=:k}
                \end{align}
                \item We use similar reasoning to figure out that the expert will choose $I$ iff
                \begin{align}
                    \boxed{k \leq \theta \leq k'}
                \end{align}
                
            \end{itemize}
        \end{solution}
        
        \item (2 points) Assume the expert chooses
        \[
        m = \begin{cases}
        G, & \theta < k \\
        I, & k \leq \theta < k' \\
        B, & \theta \geq k'
        \end{cases}
        \]
        where $0 < k < k' < 1$. What is the consistent belief $E(\theta \mid m)$ and what is the politician’s optimal policy, $x_m$, for each $m \in \{G, I, B\}$?
    \begin{solution}
        The politician is going to choose
        \[x_m = \max_{x\in [0,1]}\bbE[-(x - \theta)^2 \mid m] \iff \bbE[-2(x_m - \theta)\mid m] = 0 \iff x_m = \bbE[\theta \mid m].\]
        Thus, from part (a), we have that $m = B$ iff $\theta \geq k$ and thus 
        \[x_B = \bbE[\theta \mid B]\iff \bbE[\theta \mid \theta \geq k'] = \frac{1 + k'   }{2}\]
        Similarly, we find that
        \[x_G = \bbE[\theta \mid \theta < k] = \frac{k}{2}\]
        \[x_I = \bbE[\theta \mid k \leq \theta < k'] = \frac{k + k'}{2}\]
        \[x_B = \bbE[\theta \mid \theta \geq k'] = \frac{1 +  k'}{2}\]
    \end{solution}

        \item (2 points) Prove that when $b < \frac{1}{12}$, there exists a PBE in which the politician chooses $x_m$ for each $m \in M$ such that $0 < x_G < x_I < x_B < 1$.
    \begin{solution}
We have that 
\[k = \frac{x_I + x_G}{2} - b\]
\[k' = \frac{x_I + x_B}{2} - b\]
\[x_G = \frac{k}{2}\]
\[x_I = \frac{k + k'}{2}\]
\[x_B = \frac{1 + k'}{2}\]
We can put this into RREF to solve quickly (using wolfram alpha to simplify):
\[\left(\begin{array}{ccccc|c}
    \frac{1}{2} & \frac{1}{2} & 0 & -1 & 0 &b\\
    0 & \frac{1}{2} & \frac{1}{2} & 0 & -1 & b\\
    1 & 0 & 0 & \frac{-1}{2} & 0 & 0\\
    0 & 1 & 0 & -\frac{1}{2} & -\frac{1}{2} & 0\\
    0 & 0 & 1 & 0 & -\frac{1}{2} & \frac{1}{2}
\end{array}\right) \to \left(\begin{array}{ccccc|c}
1 & 0 & 0 & 0 & 0 & \frac{1}{6} - 2b \\
0 & 1 & 0 & 0 & 0 & \frac{1}{2} - 4b \\
0 & 0 & 1 & 0 & 0 & \frac{5}{6} - 2b \\
0 & 0 & 0 & 1 & 0 & \frac{1}{3} - 4b \\
0 & 0 & 0 & 0 & 1 & \frac{2}{3} - 4b
\end{array}\right)\] and thus 
\begin{align}
    x_G &= \frac{1}{6} - 2b\\
    x_I &= \frac{1}{2} - 4b\\
    x_B &= \frac{5}{6} - 2b\\
    k &= \frac{1}{3} - 4b\\
    k' &= \frac{2}{3} - 4b
\end{align}
Since $b < \frac{1}{12},$ we have that 
\[x_G = \frac{1}{6} - 2b > \frac{1}{6} - \frac{1}{6} = 0\]
\[x_G < x_I \iff x_G - x_I < 0 \iff -\frac{1}{3} + 2b < 0\] which is satisfied since 
\[-\frac{1}{3} + 2b < -\frac{1}{3} + \frac{1}{6} < 0\]
Moreover, 
\[x_I < x_B \iff x_I - x_B < 0 \iff -\frac{1}{3} - 2b <0 \iff -\frac{1}{3} < 2b\] which is satisfied since $b>0.$ We also have that 
\[x_B < 1 \iff \frac{1}{2} - 4b <1 \iff -\frac{1}{2} -4b <0 \iff -\frac{1}{2} < 4b \] which is satisfied since $b>0.$ Finally, we need to check that $0  < k < k' < 1.$ We have that 
\[k = \frac{1}{3} - 4b >\frac{1}{3} - 4\frac{1}{12} = 0\]
\[k < k' \iff \frac{1}{3} - 4b < \frac{2}{3} - 4b \iff -\frac{1}{3} <0\] which is satisfied.
\[1 > k'  \iff 1>\frac{2}{3} - 4b \iff \frac{1}{3} > 4b \iff \frac{1}{12} >b,\] which is clearly satisfied. 

    \end{solution}

        \item (2 points) Suppose the condition in 1.3 holds, do the politician prefer the game with $M = \{G, I, B\}$ or that with $M = \{G, B\}$? What about the expert? Explain your answer.
    \begin{solution}
For the politician, we use the law of total expectation,
\begin{align*}
    \bbE[U_P] &= \bbE[U_P \mid \theta < k]\bbP\{\theta < k\} + \bbE[U_P \mid k \leq \theta \leq k']\bbP\{k \leq \theta < k'\} + \bbE[U_P\mid \theta \geq k']\bbP\{\theta \geq k'\}\\
    &= \int_0^k (x_G- \theta)^2 \,d\theta + \int_{k}^{k'}(x_I - \theta)^2 \,d\theta + \int_{k'}^1 (x_B - \theta)^2 \,d\theta
\end{align*}
Similarly, for the expert, we see that
\begin{align*}
    \bbE[U_E] &= \int_0^k (x_G- (\theta + b))^2 \,d\theta + \int_{k}^{k'}(x_I - (\theta + b))^2 \,d\theta + \int_{k'}^1 (x_B - (\theta + b))^2 \,d\theta
\end{align*}
We use software (Python, let me know if you need the code, but its just solving the above equations using (4) - (8) above) to compare the two games (Figure 1 below) and note that for $b<\frac{1}{12} \approx 0.08,$ both the politician and the expert expect higher payoffs. Thus, this new game is more beneficial to both of them. 


    \end{solution}

    \section*{Question 2 (12 points)}
             Consider an incumbent and a decisive citizen. The incumbent may generate two possible outcomes: $y \in \{0, 1\}$, where $y = 1$ is good and $y = 0$ is bad. The citizen gets the payoff of 1 from a good outcome ($y = 1$) and 0 from a bad outcome ($y = 0$).

        The incumbent chooses how much effort to exert, $e \in [0, 1]$, in generating a good outcome. The probability of getting a good outcome is
        \[
        \Pr(y = 1 \mid \theta, e) = \theta + (1-\theta)e
        \]
        where \( \theta \in \{0, 1\} \) is the incumbent’s type. 
\( \theta = 1 \) is a competent type who always generates a good outcome with certainty regardless of the effort. 
\( \theta = 0 \) is an incompetent type whose probability to generate a good outcome equals the effort \( e \). 
Type is the incumbent’s private information and the citizen has a prior that 
$
\mathbb{P}(\theta = 1) = \frac{1}{2}.
$

The incumbent gets the reward of \( r \in (0, 1) \) from holding office. But effort is costly. 
By exerting the effort of \( e \), the incumbent suffers a cost of $\frac{e^2}{2}.$

The citizen observes the outcome \( y \) the incumbent produces and then chooses whether 
to retain or remove the incumbent in an election.
    \begin{enumerate}


        \item (1 point)
        How much effort would the incumbent exert depending on her type? What is the probability of hitting a good outcome $y = 1$ in equilibrium?
\begin{solution}
\begin{itemize}
    \item Suppose $\theta = 1$ and the incumbent is competent. Then for any $e\in [0,1],$ $\bbP\{y = 1 \mid 1,e\} = 1,$ and so the incumbent gets a payoff of $r - \frac{e^2}{2}.$ Maximizing with respect to $e,$ the incumbent exerts 
    \[\boxed{e = 0}.\]
    \item Suppose $\theta = 0$ and thus the incumbent is incompetent. Then the incumbent gets a payoff of 
    \[(\bbP\{y = 1 \mid 0, e\})(r - \frac{e^2}{2}) + (\bbP\{y = 0 \mid 0, e\})(-\frac{e^2}{2}) = (e)(r - \frac{e^2}{2}) + (1-e)(-\frac{e^2}{2})= re - \frac{e^2}{2}.\] Thus, the incumbent will maximize his payoff and so 
    \[0 = \frac{\partial }{\partial e}= r -e \implies e^* = r.\] Thus, the incumbent will exert 
    \[\boxed{e =r}\]

    Thus, 
    \[\bbP\{y =1 \} = \frac{1}{2}\bbP\{y =1 \mid \theta = 0\} + \frac{1}{2}\bbP\{y = 1 \mid \theta = 1\} = \boxed{\frac{1}{2}(r + 1)}\]
\end{itemize}
\end{solution}

        \item (2 points) Now assume in the election, a challenger campaigns against the incumbent. The challenger can also be competent or incompetent. Her type $\theta' \in \{0, 1\}$ is her private information and the citizen’s prior is also $\Pr(\theta' = 1) = \frac{1}{2}$.
        The campaign generates a signal \( s \in [0, 1] \) about \( \theta' \) such that

\[
\Pr(s \mid \theta' = 1) = 2s
\]

\[
\Pr(s \mid \theta' = 0) = 2(1 - s).
\]
        
        
        Observing the signal $s$, what is the citizen’s consistent belief about $\theta'$, $\Pr(\theta' = 1 \mid s)$? Does this depend on $y$ and why?
\begin{solution}
    Using Bayes' rule:
    \begin{align*}
        \bbP\{\theta ' = 1\mid s\} &= \frac{\bbP\{\theta' = 1, s\}}{\bbP\{s\}}\\
        &= \frac{\bbP\{\theta' = 1\}\bbP\{s \mid \theta' = 1\}}{\bbP\{s \mid \theta' = 1\}\bbP\{\theta' = 1\} + \bbP\{s \mid \theta' = 0\}\bbP\{\theta' = 0\}}\\
        &= \frac{\frac{1}{2}2s}{\frac{1}{2}2s + 2(1-s)\frac{1}{2}}\\
        &= \frac{s}{s + (1-s)}\\
        &= s
    \end{align*}
    Thus, 
    \[\boxed{\bbP\{\theta' = 1 \mid s\} = s}\] which does not depend on $y.$ The reason the signal doesn't depend on $y$ is because the citizen receives the signal before they observe $y$ because the challenger has not had a chance to be in office yet. 
\end{solution}
        \item (2 points) After the election, the office holder chosen by the citizen chooses how much effort to exert, $e'$, and generate the outcome of $y' \in \{0, 1\}$. How much effort would the office holder in the second period exert? What is the probability of hitting a good outcome $y' = 1$ depending on the office holder’s type? Given
 your answer to this question and the question in 2.2, what is the citizen’s expected payoff
 by voting the challenger into office conditional on $s$?
\begin{solution}
    There is no incentive to exert effort since there is no possibility of re-election. Thus, $e' = 0.$ Hence 
    \[\boxed{\bbP\{y' = 1\mid \theta' = 1\}= 1 \qquad \bbP\{y' = 1\mid \theta' = 0\} = 0}.\]

    The expected payoff for voting the challenger into office is 
    \begin{align*}
        \boxed{\bbP\{y' = 1\mid s\}(1) + \bbP\{y' = 0 \mid s\}(0) = \bbP\{\theta' = 1 \mid s\} = s }
    \end{align*}
\end{solution}
        \item (1 point) First prove that the competent type of the incumbent would always choose $e = 0$. Now suppose the incompetent type chooses some $e \in (0, 1)$. What is the citizen’s consistent belief about $\theta$ given $y$? Calculate $\Pr(\theta = 1 \mid y = 1)$ and $\Pr(\theta = 1 \mid y = 0)$. Given your answer and that to 2.3, when would the citizen retain the incumbent given $y$ and
 $s$?
        \begin{solution}
            We showed last problem that $e' = 0$ in the second period. We showed in the first problem that $e = 0$ in the first period. Thus, $e = 0$ always for $\theta = 1.$ Computing the beliefs, we have that 
            \begin{align*}
                \bbP\{\theta = 1 \mid y = 1\} &= \frac{\bbP\{y = 1 \mid \theta = 1\}\bbP\{\theta = 1\}}{\bbP\{y = 1 \mid \theta = 1\}\bbP\{\theta = 1\} + \bbP\{y = 1 \mid \theta = 0\}\bbP\{\theta = 0\}}\\
                &= \frac{(1) \frac{1}{2}}{(1)\frac{1}{2} + e \frac{1}{2}}\\
                &= \boxed{\frac{1}{1 + e}}
            \end{align*}
            where the last equality follows from the result in part (a). Similarly,
            \begin{align*}
                \bbP\{\theta = 1 \mid y  =0\} &= \frac{\bbP\{y = 0 \mid \theta = 1\}\bbP\{\theta= 1\}}{\bbP\{y = 0 \mid \theta = 1\}\bbP\{\theta= 1\} + \bbP\{y = 0 \mid \theta = 0\}\bbP\{\theta = 0\}}\\
                &= \frac{0}{c} = \boxed{0}
            \end{align*}
            The citizen will retain the incumbent if the expected payoff of the incumbent's second term is higher than the expected payoff of the challenger, which we found to be $s$ in the above problem. Thus, we re-elect the incumbent iff
            \begin{align*}
            s &\leq \bbP\{\theta = 1 \mid y\}(1) + \bbP\{\theta = 0 \mid y\}(0)\\ &= \bbP\{\theta = 1 \mid y\}\\
            &= \begin{cases}
                \frac{1}{1 + e}\qquad y = 1\\
                0 \qquad \,\:\:\:\;y = 0
            \end{cases}
            \end{align*}
            and so the we re-elect iff either
            \[\boxed{y = 1\text{ and } s\leq \frac{1}{1 + e}}\]
            or 
            \[\boxed{y = 0 \text{ and } s = 0}\]
        \end{solution}

        \item (2 points) Consider the incompetent type of the incumbent in the first period. Suppose she chooses the effort of $e \in (0, 1)$. What is her probability of getting retained?
    \begin{solution}
\begin{lemma}
    $s$ is uniformly distributed on $[0,1]$
\end{lemma}
\begin{proof}
    Let $x \in [0,1].$ We have that

    \begin{align*}
        F_s(x) &= \bbP\{s \leq x\}\\
        &= \bbP\{s \leq x \mid \theta' = 1\}\bbP\{\theta' = 1\} + \bbP\{s \leq x \mid \theta' = 0 \}\bbP\{\theta' = 0\}\\
        &= \int_0^x (2s)\frac{1}{2}\,ds + \int_0^x 2(1-s)\frac{1}{2}\,ds\\
        &= \int_0^x s\,ds + \int_0^x (1-s)\,ds\\
        &= \int_0^x 1 \,ds\\
        &= x
    \end{align*}
\end{proof}

        We have that if $\theta = 0,$ we have computed that
        \[\bbP\{y = 1\} = e\qquad \bbP\{y = 0 \} = (1-e)\] and thus 
        from the previous problem, 
        \begin{align*}
            \bbP\{y= 1, s \leq \frac{1}{1 + e}\} &= \bbP\{s \leq \frac{1}{1 + e} \mid y = 1\}\bbP\{y = 1\}\\
            &= \bbP\{s \leq \frac{1}{1 + e}\}\bbP\{y = 1\}\\
            &= \frac{e}{1 + e}
        \end{align*}
        Where we use the fact from 2.2 that $s$ is independent from $y.$  Moreover, 
        \begin{align*}
            \bbP\{y = 1 , s = 0\} &= \bbP\{y= 1\}\bbP\{s = 0\} =0
        \end{align*}  since $s$ is uniform. Hence, we have by 2.4 that 
        \[\bbP\{\text{re-election}\} = \bbP\{y= 1, s \leq \frac{1}{1 + e}\} + \bbP\{y = 1 , s = 0\} = \boxed{\frac{e}{1  +e} }\]
    \end{solution}

        \item (2 points) What is the optimal effort for the incompetent type? Derive the equation that identifies this optimal effort.
    \begin{solution}
        Let $\theta = 0.$ We know that the expected payoff if 
    \begin{align*}
    E_e = \bbE[u \mid e, \theta = 0] &= \bbP\{\text{re-election}\}(r - \frac{e^2}{2}) + \bbP\{\text{ousted}\}(-\frac{e^2}{2})\\
        &= \frac{e}{1 + e}(r - \frac{e^2}{2}) + \frac{1}{1 + e}(-\frac{e^2}{2})\\
        &= \frac{1}{1 + e}(e(r - \frac{e^2}{2}) - \frac{e^2}{2})\\
        &= \frac{1}{1 + e}\left(er - \frac{e^3}{2} - \frac{e^2}{2}\right)
    \end{align*}
    Differentiating, 
    \[0 = \frac{\partial E_e}{\partial e} = \frac{r - e(1 + e)^2}{(1 + e)^2}\] This only happens when the numerator vanishes, and thus $e^*$ solves 
    \[\boxed{e^*(1 + e^*)^2 = r}\]
    \end{solution}

        \item (2 points) Calculate the equilibrium probability of hitting a good outcome $y = 1$ in the first period. Prove that this probability is lower than the probability of $y = 1$ in 2.1. Comment on this result.
    \begin{solution}
Calculating, 
\begin{align*}
    \bbP\{y = 1\} &= \bbP\{y = 1 \mid \theta = 1\}\bbP\{\theta = 1\} + \bbP\{y = 1\mid \theta = 0\}\bbP\{\theta = 0\}\\
    &= \frac{1}{2} + \frac{1}{2}e^*
\end{align*} Without a challenger, we know that 
\[\bbP\{y = 1\} = \frac{1}{2} + \frac{1}{2}r.\] But we know that 
\[r = e^* (1 + e^*)^2 \geq e^*,\] and thus 
\[\frac{1}{2}  + \frac{1}{2}r \geq \frac{1}{2} + \frac{1}{2}e^*,\] and the conclusion follows. Hence, the probability that the citizens get a good outcome in the first period is lower if there is a challenger. This lowkey sucks-democracy yields a higher probability of worse results. 
    \end{solution}
    \end{enumerate}
