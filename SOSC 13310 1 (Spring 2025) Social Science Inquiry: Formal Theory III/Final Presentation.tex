\section{Introduction}
\begin{frame}{Setup}
    \begin{itemize}
        \item Creon rules over a fairly unhappy population as a dictator. The kingdom he inherited from Oedipus is tired of being ruled by Oedipus' family `tree'.
        \item Antigone decides to break one of Creon's laws. Namely she follows what she believes to be the natural law and gives burial rites to her brother (who might've also been her lover!)

    \end{itemize}

% Slide 2
\section{Players and Actions}
\begin{frame}{Players}
    \begin{itemize}
        \item A dictator $D$ has the option to punish $p\in \{0,1\},$ where $p =1 $ indicates punishment. The dictator has strength $\theta$
        \item A continuum of subjects, $i \in [0,1],$ can either rebel $a_i = 1$ or cower $a_i = 0.$ Rebelling costs $c>0$ and winning the rebellion is rewarded with $r>0.$ We assume $r>c.$
    \end{itemize}
\end{frame}

\section{The Game}
\begin{frame}{The Game and Information Structure}
\begin{itemize}
    \item Nature awards $D$ some strength $\theta.$ 
    \item $D$ receives a private signal $s$ about $\theta.$ $D$ decides to punish or not. If $p = 1,$ the Gods will see this and kill him with a probability $\mu.$
    \item Each citizen $i$ observes $p,$ but they do not observe whether the Gods have killed $D,$ but the citizens do believe in the Gods. Each $i$ observes a private signal about $\theta.$ and Each $i$ will choose to rebel or not depending on $p.$ 
    \item If the Gods did not kill $D,$ then the rebellion will kill him if and only if 
    \[\int_0^1 a_i \,di \geq \theta\]
\end{itemize}
    
\end{frame}

\begin{frame}{Payoff Structure}
\begin{itemize}
    \item For each $i,$ the payoffs are given by:
    \[\begin{tabular}{c|c|c}
            micro/macro & win  & lose\\
             \hline
             rebel ($a_i = 1$)& $r -c$ & $-c$ \\
             \hline
             cower ($a_i = 0$)&  $p \cdot r$& $0$\\
        \end{tabular}\] Note how the citizens faith in god plays in: If $p = 1,$ they believe that not rebelling will still yield them a reward since the Gods will do their work for them. 
    \item The dictator gets $1$ util from staying in power and $0$ util from being deposed/killed.
\end{itemize}
    
\end{frame}

% Slide 4
\section{Information Structure}
\begin{frame}{Information Structure}
Let $K$ represent the event $D$ gets ousted/killed
    \begin{itemize}
        \item $D$ has a strength $\theta \in [0,1]$. Everybody knows $\theta$ has the prior that
        \[\theta \sim \text{Uniform}(0,1)\] The subjects receive a signal generated by
        \[s_i = \theta + \varepsilon_i\] where \[\varepsilon_i\sim F \text{ i.i.d.}\] 
        and $D$ receives a signal generated by 
        \[s = \theta + \varepsilon,\] where $\varepsilon$ is drawn from $F.$
        \item The gods kill with probability $\mu$ given $p = 1.$  Thus, 
        \[\bbP\{K \mid p = 1\} = \mu\] independent of $\theta.$
    \end{itemize}
\end{frame}

\section{PBE Candidate}
\begin{frame}{Solution}
\begin{itemize}
    \item Suppose $p= 0,$ then 
\[\bbE[a_i = 1] = \bbP\{K \mid p = 1\}(r-c) + (1-\bbP\{K\mid p=1\}) =\mu r - c\]
\[\bbE[a_i = 0]= \bbP\{K \mid p =1\}r + 0 = \mu r.\] Hence, if $p = 0,$ then the citizens will not rebel no matter what. 
\item Suppose $p = 1,$ then 
\[\bbE[a_1 = 0] = 0\] and 
\begin{align*}
    \bbE[a_1 = 1] &= \bbP\{K \mid p = 1, s_i\}(r-c) + (1-\bbP\{K\mid p=1, s_i\})\\
    &= r \frac{\int_0^{\theta^*}\left[1 - F(s^* - \theta)\right]f_{\varepsilon_i}(s_i - \theta)\,d\theta}{\int_0^1\left[1 - F(s^* - \theta)\right]f_{\varepsilon_i}(s_i - \theta)\,d\theta}-c
\end{align*}

\end{itemize}
\end{frame}

\begin{frame}{Calculating}
We know that $K$ happens iff $\theta \leq \theta^*$ for some $\theta^*.$ It is also not hard to see that $p$ and $s_i$ are conditionally independent since $s_i$ is drawn independently of $p.$ Moreover, we know that $p = 0$ if and only if $s\geq s^*$ for some critical value $s^*.$
\begin{align*}
    \bbP\{\theta \mid p = 0, s_i\} &= \frac{\bbP\{p = 0\mid \theta\}\bbP\{s_i \mid \theta\}\bbP\{\theta\}}{\int_0^1\bbP\{p = 0 \mid \theta\}\bbP\{s_i \mid \theta\}\,d\theta}\\
    &= \frac{\bbP\{s \geq s^* \mid \theta\}\bbP\{s_i = \theta + \varepsilon_i\}}{\int_0^1 \bbP\{s \geq s^* \mid \theta\}\bbP\{s_i = \theta + \varepsilon_i\}\,d\theta}\\
    &= \frac{\bbP\{\theta  + \varepsilon \geq s^*\}f_{\varepsilon_i}(s_i - \theta)}{\int_0^1\bbP\{\theta  + \varepsilon \geq s^*\}f_{\varepsilon_i}(s_i - \theta)\,d\theta}\\
    &= \frac{\left[1 - F(s^* - \theta)\right]f_{\varepsilon_i}(s_i - \theta)}{\int_0^1\left[1 - F(s^* - \theta)\right]f_{\varepsilon_i}(s_i - \theta)\,d\theta}
\end{align*}
\end{frame}
\begin{frame}{Deriving $k$ and $\theta^*$}
\begin{itemize}
    \item $k$ happens when $i$ is indifferent. That is, $k$ is the unique number such that:
    \begin{align}
    \frac{\int_0^{\theta^*}\left[1 - F(s^* - \theta)\right]f_{\varepsilon_i}(k - \theta)\,d\theta}{\int_0^1\left[1 - F(s^* - \theta)\right]f_{\varepsilon_i}(k - \theta)\,d\theta} = \frac{c}{r}    
    \end{align}
    
    \item From class, we know that $\theta^*$ is such that 
    \begin{align}
    \theta^* = F(k - \theta^*)    
    \end{align}
    
\end{itemize}
\end{frame}


\begin{frame}{Dictator cutoff}
\begin{itemize}
    \item For the dictator:
    \[\bbE[p= 1] = \bbP\{K\} 0 + (1-\bbP\{K\})1 = 1- \mu\]
    \[\bbE[p = 0] = (1-\bbP\{K\}) = (1- \bbP\{\theta \leq \theta^*\mid s\})= 1 - F(\theta^* - s)\]
    \item Hence, the dictator will punish if and only if 
    \[1-F(s - \theta^*)\geq \mu\] and will be indifferent when 
    \[1-F(\theta^* - s^*) = \mu\] or 
    \begin{align}
        \theta^* - s^*  = F^{-1}(\theta^* - s^*)
    \end{align}
Thus, solving equations (1), (2), (3) gives an equilibrium strategy.
\end{itemize}

\end{frame}
\begin{frame}{Solution}
    \begin{itemize}
        \item The dictator should punish iff 
        \[s \leq s^*\]
        \item If $p = 0,$ the citizens should not attack.
        \item If $p = 1,$ the citizens will rebel if and only if
        \[s_i \leq k\]
    \end{itemize}
We calculate $s,k,$ and $\theta$ using software.
\end{frame}
